cmake_minimum_required(VERSION 3.16)
project(dis_xs_kc VERSION 0.1 LANGUAGES C CXX)

# C++ implementation of the kinematically corrected (finite-energy) DIS cross
# sections of src/small_x_physics. See README.md in this directory.

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if (NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Release")
endif()
message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")

option(unittest "Build unit tests" OFF)
option(native "Optimize for the CPU of the build machine (-march=native / -mcpu=native)" ON)

# No -ffast-math: the quadrupole relies on IEEE NaN semantics (sqrt of a
# negative discriminant must compare false against 0), exactly as numpy does.
string(APPEND CMAKE_CXX_FLAGS " -Wall -Wextra -Wshadow")
if (native)
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-march=native" HAS_MARCH_NATIVE)
    check_cxx_compiler_flag("-mcpu=native" HAS_MCPU_NATIVE)
    if (HAS_MARCH_NATIVE)
        string(APPEND CMAKE_CXX_FLAGS " -march=native")
    elseif (HAS_MCPU_NATIVE)
        string(APPEND CMAKE_CXX_FLAGS " -mcpu=native")
    endif()
endif()

find_package(GSL REQUIRED)

add_subdirectory(external/cuba-4.2.2)
add_subdirectory(src)

# Doctest unit tests (tests/), compared against reference values generated by
# the Python implementation (tests/reference/). Off by default; enable with
# -Dunittest=ON.
if (unittest)
    enable_testing()
    add_subdirectory(tests)
endif()
